    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const codeWrappers = document.querySelectorAll('.code-wrapper');
            
            codeWrappers.forEach(wrapper => {
                const codeElement = wrapper.querySelector('code');
                const outputDiv = wrapper.querySelector('.output');
                const codeText = codeElement.innerText;

                // --- Botón Copiar ---
                const copyBtn = document.createElement('button');
                copyBtn.className = 'btn copy-btn';
                copyBtn.textContent = 'Copiar';
                copyBtn.addEventListener('click', async () => {
                    try {
                        await navigator.clipboard.writeText(codeText);
                        copyBtn.textContent = '¡Copiado!';
                        copyBtn.classList.add('copied');
                        setTimeout(() => {
                            copyBtn.textContent = 'Copiar';
                            copyBtn.classList.remove('copied');
                        }, 2000);
                    } catch (err) {
                        console.error('Error al copiar: ', err);
                    }
                });

                // --- Botón Ejecutar ---
                const execBtn = document.createElement('button');
                execBtn.className = 'btn exec-btn';
                execBtn.textContent = 'Ejecutar';
                execBtn.addEventListener('click', async () => {
                    if (!confirm('¿Estás seguro de que deseas ejecutar este script en la base de datos?')) return;

                    execBtn.disabled = true;
                    execBtn.textContent = 'Ejecutando...';
                    outputDiv.className = 'output';
                    outputDiv.textContent = '';

                    try {
                        const response = await fetch('/api/ejecutar', {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify({ script: codeText })
                        });

                        const data = await response.json();

                        if (data.success) {
                            outputDiv.className = 'output success';
                            outputDiv.textContent = '✅ ' + data.message;
                            if (data.results && data.results.length > 0) {
                                outputDiv.textContent += '\n' + JSON.stringify(data.results, null, 2);
                            }
                        } else {
                            outputDiv.className = 'output error';
                            outputDiv.textContent = '❌ Error: ' + data.error;
                        }
                    } catch (error) {
                        outputDiv.className = 'output error';
                        outputDiv.textContent = '❌ Error de conexión con el servidor.';
                    } finally {
                        execBtn.disabled = false;
                        execBtn.textContent = 'Ejecutar';
                    }
                });

                wrapper.appendChild(copyBtn);
                wrapper.appendChild(execBtn);
            });
        });
    </script>
